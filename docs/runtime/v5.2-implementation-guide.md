### 2.2 バリデーションスクリプト

`scripts/validate_event.py` を作成：

```python
#!/usr/bin/env python3
"""イベントレコードのバリデーション"""

import json
import sys
from pathlib import Path
import jsonschema

def load_schema(schema_path: Path) -> dict:
    """JSON Schema を読み込む"""
    with open(schema_path, encoding='utf-8') as f:
        return json.load(f)

def validate_event(event: dict, schema: dict) -> list[str]:
    """イベントレコードを検証"""
    errors = []
    validator = jsonschema.Draft7Validator(schema)
    
    for error in validator.iter_errors(event):
        errors.append(f"{'.'.join(map(str, error.path))}: {error.message}")
    
    return errors

def validate_jsonl_file(file_path: Path, schema: dict) -> tuple[int, int]:
    """JSONL ファイル全体を検証"""
    valid_count = 0
    error_count = 0
    
    with open(file_path, encoding='utf-8') as f:
        for line_num, line in enumerate(f, 1):
            line = line.strip()
            if not line:
                continue
            
            try:
                event = json.loads(line)
                errors = validate_event(event, schema)
                
                if errors:
                    error_count += 1
                    print(f"❌ {file_path.name}:{line_num}")
                    for error in errors:
                        print(f"   - {error}")
                else:
                    valid_count += 1
                    
            except json.JSONDecodeError as e:
                error_count += 1
                print(f"❌ {file_path.name}:{line_num} - JSON parse error: {e}")
    
    return valid_count, error_count

if __name__ == "__main__":
    schema_path = Path("schemas/event.schema.json")
    schema = load_schema(schema_path)
    
    events_dir = Path("work/events")
    if not events_dir.exists():
        print("work/events/ ディレクトリが存在しません")
        sys.exit(1)
    
    total_valid = 0
    total_errors = 0
    
    for jsonl_file in events_dir.glob("*.jsonl"):
        valid, errors = validate_jsonl_file(jsonl_file, schema)
        total_valid += valid
        total_errors += errors
        
        if errors == 0:
            print(f"✅ {jsonl_file.name}: {valid} events valid")
    
    print(f"\n合計: {total_valid} valid, {total_errors} errors")
```

## Phase 3: Extraction Agent の実装

### 3.1 LLM クライアントの設定

`scripts/llm_client.py` を作成：

```python
#!/usr/bin/env python3
"""Qwen3.8-27B との通信クライアント"""

from openai import OpenAI
import yaml
from pathlib import Path

def load_config() -> dict:
    """設定ファイルを読み込む"""
    config_path = Path("config.yaml")
    with open(config_path, encoding='utf-8') as f:
        return yaml.safe_load(f)

def create_client(config: dict) -> OpenAI:
    """OpenAI 互換クライアントを作成"""
    return OpenAI(
        base_url=config['llm']['base_url'],
        api_key=config['llm']['api_key']
    )

def extract_events(client: OpenAI, model: str, chapter: str, raw_text: str) -> list[dict]:
    """原文からイベントを抽出"""
    
    system_prompt = """
あなたは「幻想再帰のアリュージョニスト」の事実抽出エージェントです。

【鉄則】
1. 原文にない情報は絶対に出さない
2. 各事実に証拠（quote）を付与
3. 出力は必ずJSON配列形式

【出力形式】
[
  {
    "event_id": "E_ch{NNNN}_{entity_type}_{entity}_{aspect}_{observation_type}",
    "episode": "ch{NNNN}",
    "entity": "エンティティ名",
    "entity_type": "character|terminology|organization|item|motif|relationship|phrase",
    "aspect": "visual|name|speech|action|relationship|symbolic",
    "observation": "観察内容",
    "quote": "原文からの引用",
    "locator": {
      "chapter": "ch{NNNN}",
      "lines": "行番号（推定で可）"
    },
    "spoiler_after": "ch{NNNN}"
  }
]

【aspect の略称】
- visual → V
- name → N
- speech → S
- action → A
- relationship → R
- symbolic → Y
"""

    response = client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": f"以下の原文（{chapter}）からイベントを抽出してください：\n\n{raw_text}"}
        ],
        temperature=0.1,
        response_format={"type": "json_object"}
    )
    
    # JSON をパース
    import json
    result = json.loads(response.choices[0].message.content)
    
    # JSON配列として取得
    if isinstance(result, dict) and 'events' in result:
        return result['events']
    elif isinstance(result, list):
        return result
    else:
        return [result]
```

### 3.2 メイン処理スクリプト

`scripts/extract_chapter.py` を作成：

```python
#!/usr/bin/env python3
"""1章分の原文からイベントを抽出"""

import json
import sys
from pathlib import Path
from datetime import date
import yaml

from llm_client import load_config, create_client, extract_events
from validate_event import load_schema, validate_event

def process_chapter(chapter_num: int, config: dict, client, schema: dict) -> bool:
    """1章を処理"""
    
    chapter = f"ch{chapter_num:04d}"
    raw_path = Path(config['paths']['raw']) / f"{chapter}.txt"
    
    if not raw_path.exists():
        print(f"⚠️  {raw_path} が存在しません")
        return False
    
    print(f"\n📖 {chapter} を処理中...")
    
    # 原文を読み込み
    raw_text = raw_path.read_text(encoding='utf-8')
    
    # イベントを抽出
    events = extract_events(
        client,
        config['llm']['model'],
        chapter,
        raw_text
    )
    
    print(f"   {len(events)} 件のイベントを抽出")
    
    # バリデーション
    valid_events = []
    invalid_count = 0
    
    for event in events:
        # created フィールドを追加
        if 'created' not in event:
            event['created'] = date.today().isoformat()
        
        errors = validate_event(event, schema)
        if errors:
            invalid_count += 1
            print(f"   ❌ Invalid event: {errors}")
        else:
            valid_events.append(event)
    
    # イベント層に書き込み
    output_path = Path(config['paths']['work']) / 'events' / f"{chapter}.jsonl"
    output_path.parent.mkdir(parents=True, exist_ok=True)
    
    with open(output_path, 'w', encoding='utf-8') as f:
        for event in valid_events:
            f.write(json.dumps(event, ensure_ascii=False) + '\n')
    
    print(f"   ✅ {output_path} に {len(valid_events)} 件書き込み")
    
    if invalid_count > 0:
        print(f"   ⚠️  {invalid_count} 件が無効（スキップ）")
    
    return True

if __name__ == "__main__":
    config = load_config()
    client = create_client(config)
    schema = load_schema(Path(config['paths']['schemas']) / 'event.schema.json')
    
    # 処理範囲
    start = config['chapter_range']['start']
    end = config['chapter_range']['end']
    
    success_count = 0
    for chapter_num in range(start, end + 1):
        if process_chapter(chapter_num, config, client, schema):
            success_count += 1
    
    print(f"\n完了: {success_count}/{end - start + 1} 章を処理")
```

## Phase 4: 集約と正規化

### 4.1 集約スクリプト

`scripts/aggregate_events.py` を作成：

```python
#!/usr/bin/env python3
"""イベント層から中間集約層を生成"""

import json
import yaml
from pathlib import Path
from collections import defaultdict
from datetime import date

def aggregate_chapter(chapter: str, events_dir: Path, staging_dir: Path) -> bool:
    """1章分のイベントを集約"""
    
    events_file = events_dir / f"{chapter}.jsonl"
    if not events_file.exists():
        return False
    
    # イベントを読み込み
    events = []
    with open(events_file, encoding='utf-8') as f:
        for line in f:
            line = line.strip()
            if line:
                events.append(json.loads(line))
    
    # エンティティごとに集約
    entities = defaultdict(lambda: {
        'entity_type': None,
        'aspects': defaultdict(list),
        'first_appearance_candidate': False
    })
    
    for event in events:
        entity_key = event['entity']
        entities[entity_key]['entity_type'] = event['entity_type']
        entities[entity_key]['aspects'][event['aspect']].append({
            'event_ref': event['event_id'],
            'quote': event.get('quote', ''),
            'locator': event.get('locator', {}),
            'spoiler_after': event.get('spoiler_after', chapter)
        })
    
    # YAML 形式に変換
    staging_data = {
        'episode': chapter,
        'created': date.today().isoformat(),
        'status': 'pending_merge',
        'entities': []
    }
    
    for entity_name, entity_data in entities.items():
        entity_entry = {
            'canonical_target': f"[[E_{entity_data['entity_type']}_{entity_name}]]",
            'entity_type': entity_data['entity_type'],
            'aspects': {},
            'first_appearance_candidate': True  # PoC ではすべて候補
        }
        
        for aspect, observations in entity_data['aspects'].items():
            entity_entry['aspects'][aspect] = {
                'observations': observations
            }
        
        staging_data['entities'].append(entity_entry)
    
    # 中間集約層に書き込み
    output_path = staging_dir / f"staging_{chapter}.yaml"
    output_path.parent.mkdir(parents=True, exist_ok=True)
    
    with open(output_path, 'w', encoding='utf-8') as f:
        yaml.dump(staging_data, f, allow_unicode=True, default_flow_style=False, sort_keys=False)
    
    print(f"✅ {output_path} に集約（{len(staging_data['entities'])} エンティティ）")
    return True

if __name__ == "__main__":
    events_dir = Path("work/events")
    staging_dir = Path("work/staging")
    
    for events_file in events_dir.glob("*.jsonl"):
        chapter = events_file.stem
        aggregate_chapter(chapter, events_dir, staging_dir)
```

### 5.1 SQLite による構造化検索

`scripts/build_search_index.py` を作成：

```python
#!/usr/bin/env python3
"""イベント層から SQLite 検索インデックスを構築"""

import json
import sqlite3
from pathlib import Path

def create_database(db_path: Path):
    """データベースを作成"""
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()
    
    # イベントテーブル
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS events (
            event_id TEXT PRIMARY KEY,
            episode TEXT NOT NULL,
            entity TEXT NOT NULL,
            entity_type TEXT,
            aspect TEXT,
            observation TEXT,
            quote TEXT,
            spoiler_after TEXT,
            created TEXT
        )
    ''')
    
    # インデックス
    cursor.execute('CREATE INDEX IF NOT EXISTS idx_entity ON events(entity)')
    cursor.execute('CREATE INDEX IF NOT EXISTS idx_aspect ON events(aspect)')
    cursor.execute('CREATE INDEX IF NOT EXISTS idx_episode ON events(episode)')
    
    conn.commit()
    return conn

def insert_events(conn: sqlite3.Connection, events_dir: Path):
    """イベントを挿入"""
    cursor = conn.cursor()
    
    for jsonl_file in events_dir.glob("*.jsonl"):
        with open(jsonl_file, encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                
                event = json.loads(line)
                
                cursor.execute('''
                    INSERT OR REPLACE INTO events 
                    (event_id, episode, entity, entity_type, aspect, observation, quote, spoiler_after, created)
                    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
                ''', (
                    event['event_id'],
                    event['episode'],
                    event['entity'],
                    event.get('entity_type'),
                    event.get('aspect'),
                    event.get('observation'),
                    event.get('quote'),
                    event.get('spoiler_after'),
                    event.get('created')
                ))
    
    conn.commit()

def query_events(conn: sqlite3.Connection, entity: str = None, aspect: str = None, episode: str = None):
    """イベントを検索"""
    cursor = conn.cursor()
    
    query = "SELECT * FROM events WHERE 1=1"
    params = []
    
    if entity:
        query += " AND entity = ?"
        params.append(entity)
    
    if aspect:
        query += " AND aspect = ?"
        params.append(aspect)
    
    if episode:
        query += " AND episode = ?"
        params.append(episode)
    
    query += " ORDER BY episode, event_id"
    
    cursor.execute(query, params)
    return cursor.fetchall()

if __name__ == "__main__":
    db_path = Path("search/events.db")
    db_path.parent.mkdir(parents=True, exist_ok=True)
    
    # データベースを作成
    conn = create_database(db_path)
    
    # イベントを挿入
    events_dir = Path("work/events")
    insert_events(conn, events_dir)
    
    print(f"✅ {db_path} にインデックスを構築")
    
    # テストクエリ
    results = query_events(conn, entity="ハルベルト", aspect="visual")
    print(f"\nテストクエリ: ハルベルトの視覚描写")
    for row in results:
        print(f"  - {row[1]}: {row[5]}")
    
    conn.close()
```

### 5.2 クエリインターフェース

`scripts/query_cli.py` を作成：

```python
#!/usr/bin/env python3
"""検索クエリの CLI インターフェース"""

import sqlite3
from pathlib import Path
from rich.console import Console
from rich.table import Table

console = Console()

def query_interactive(conn: sqlite3.Connection):
    """インタラクティブなクエリ"""
    
    while True:
        console.print("\n[bold cyan]検索クエリ[/bold cyan]")
        console.print("  entity: エンティティ名")
        console.print("  aspect: visual|name|speech|action|relationship|symbolic")
        console.print("  episode: ch0001")
        console.print("  quit: 終了\n")
        
        entity = console.input("エンティティ名（空白でスキップ）: ").strip() or None
        aspect = console.input("アスペクト（空白でスキップ）: ").strip() or None
        episode = console.input("エピソード（空白でスキップ）: ").strip() or None
        
        if entity == 'quit' or aspect == 'quit' or episode == 'quit':
            break
        
        # クエリ実行
        cursor = conn.cursor()
        query = "SELECT episode, entity, aspect, observation, quote FROM events WHERE 1=1"
        params = []
        
        if entity:
            query += " AND entity = ?"
            params.append(entity)
        if aspect:
            query += " AND aspect = ?"
            params.append(aspect)
        if episode:
            query += " AND episode = ?"
            params.append(episode)
        
        query += " ORDER BY episode"
        
        cursor.execute(query, params)
        results = cursor.fetchall()
        
        # 結果を表示
        if results:
            table = Table(title=f"検索結果（{len(results)}件）")
            table.add_column("エピソード", style="cyan")
            table.add_column("エンティティ", style="green")
            table.add_column("アスペクト", style="yellow")
            table.add_column("観察", style="white")
            
            for row in results:
                table.add_row(row[0], row[1], row[2], row[3][:50])
            
            console.print(table)
        else:
            console.print("[yellow]該当するイベントが見つかりませんでした[/yellow]")

if __name__ == "__main__":
    db_path = Path("search/events.db")
    
    if not db_path.exists():
        console.print("[red]search/events.db が存在しません。先に build_search_index.py を実行してください。[/red]")
        exit(1)
    
    conn = sqlite3.connect(db_path)
    query_interactive(conn)
    conn.close()
```
